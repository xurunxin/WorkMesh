$taskCleanupTargets = @(

$taskCleanupResults = @()
foreach ($taskCleanupTarget in $taskCleanupTargets) {
  $taskResolvedTarget = [IO.Path]::GetFullPath($taskCleanupTarget)
  if ($taskResolvedTarget -ne $taskCleanupTarget -or -not $taskResolvedTarget.StartsWith($taskRepositoryRoot + '\', [StringComparison]::OrdinalIgnoreCase)) { throw 'target escaped task workspace' }
  if (Test-Path -LiteralPath $taskResolvedTarget) {
    $taskTargetItem = Get-Item -LiteralPath $taskResolvedTarget -Force
    if ($taskTargetItem.Attributes -band [IO.FileAttributes]::ReparsePoint) { throw 'unexpected reparse target' }
    if ($taskTargetItem.PSIsContainer -and @(Get-ChildItem -LiteralPath $taskResolvedTarget -Force -Recurse -Attributes ReparsePoint).Count -gt 0) { throw 'unexpected nested reparse point' }
    Remove-Item -LiteralPath $taskResolvedTarget -Recurse -ErrorAction Stop
  }
  $taskCleanupResults += [PSCustomObject]@{ Path=$taskResolvedTarget; VerifiedAbsent=(-not (Test-Path -LiteralPath $taskResolvedTarget)); Action='Remove-Item -LiteralPath -Recurse'; ArchiveVerified=$true }
}
$taskCleanupResults | ConvertTo-Json -Depth 4 | Set-Content -LiteralPath docs/reviews/c3/main-combination/temporary-cleanup.json -Encoding utf8
